The counter stops incrementing when it reaches the maximum value of `3` (or `2'b11`), ensuring that the output remains within the 2-bit range of values.
module top_module(
  input 	   clk,
  input 	   rst,
  input 	   en,
  output reg [1:0] data_out
);
