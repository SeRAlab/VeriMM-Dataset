The module also features a synchronous reset, ensuring that on reset it initializes the output and valid signal to zero.
module top_module(
  input clk,
  input rst_n,
  input in_valid,
  input [15:0] in_a,
  input [15:0] in_b,
  input in_mode,
  output reg out_valid,  // Changed from wire to reg

);
