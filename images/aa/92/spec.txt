If a = 1, b = 1, c = 0, d = 1, then u = 1.
If a = 0, b = 0, c = 1, d = 0, then u = 0.
If a = 1, b = 1, c = 1, d = 1, then u = 0.
If a = 0, b = 1, c = 1, d = 1, then u = 1.
If a = 1, b = 0, c = 0, d = 1, then u = 0.
If a = 0, b = 0, c = 1, d = 1, then u = 1.
If a = 1, b = 1, c = 0, d = 0, then u = 0.
If a = 0, b = 1, c = 0, d = 0, then u = 1.

module boolean_logic (
	input a,
	input b,
	input c,
	input d,
	output reg u);
