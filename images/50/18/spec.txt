Design a Verilog module for a 32-bit wide dual-port memory with synchronous read and write operations. The memory should support two independent read and write ports.
module dual_port_memory(
    input clk,
    input [4:0] read_addr1,
    input [4:0] read_addr2,
    input [4:0] write_addr1,
    input [4:0] write_addr2,
    input write_enable1,
    input write_enable2,
    input [31:0] write_data1,
    input [31:0] write_data2,
    output [31:0] read_data1,
    output [31:0] read_data2);
