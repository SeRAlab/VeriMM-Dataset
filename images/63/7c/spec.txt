Design a clock management module that takes two input clock signals and two reset signals and produces two output clock signals along with their corresponding active-low reset signals. Specifically, it directly routes the first input clock to an output for a 27 MHz clock and the second input clock to an output for a 25 MHz clock, while also transferring the associated reset signals as active-low outputs.
module top_module(
		input  wire  clk_0_clk,              
		input  wire  clk_1_clk,              
		output wire  clock_25_clk,           
		output wire  clock_25_reset_reset_n, 
		output wire  clock_27_clk_clk,       
		output wire  clock_27_reset_reset_n, 
		input  wire  reset_0_reset_n,        
		input  wire  reset_1_reset_n         
	);
