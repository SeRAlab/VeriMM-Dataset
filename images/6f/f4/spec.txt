If a = 0, b = 0, c = 1, d = 0, then z = 1.
If a = 1, b = 1, c = 0, d = 1, then z = 0.
If a = 0, b = 1, c = 1, d = 1, then z = 1.
If a = 1, b = 0, c = 0, d = 1, then z = 0.
If a = 1, b = 1, c = 1, d = 0, then z = 1.

module logic_circuit (
	input a,
	input b,
	input c,
	input d,
	output reg z);
