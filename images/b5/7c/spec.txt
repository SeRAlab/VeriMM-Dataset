If p = 1, q = 1, r = 0, s = 0, then x = 1.
If p = 0, q = 0, r = 1, s = 1, then x = 0.
If p = 1, q = 0, r = 0, s = 0, then x = 1.
If p = 0, q = 1, r = 1, s = 0, then x = 0.

module conditional_logic (
	input p,
	input q,
	input r,
	input s,
	output reg x);
