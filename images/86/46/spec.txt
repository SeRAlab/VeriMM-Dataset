If e = 0, f = 1, g = 0, h = 1, then i = 1.
If e = 1, f = 0, g = 1, h = 1, then i = 0.
If e = 1, f = 1, g = 0, h = 0, then i = 1.
If e = 0, f = 0, g = 1, h = 0, then i = 0.
If e = 1, f = 1, g = 1, h = 1, then i = 1.

module combinational_logic (
	input e,
	input f,
	input g,
	input h,
	output reg i);
