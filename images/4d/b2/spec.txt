The output will be HIGH if any of the following conditions are satisfied: both A and B are HIGH, both C is LOW and D is HIGH, E and F are different (exclusive OR), or both G is LOW and H is HIGH.
module top_module(
    input A,
    input B,
    input C,
    input D,
    input E,
    input F,
    input G,
    input H,
    output X
);
